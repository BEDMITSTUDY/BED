\chapter{Notation}

This appendix summarizes the notation used throughout the thesis.

\section*{State Variables}
\begin{tabular}{ll}
$r_t$ & Recency (time since last event) \\
$h_t$ & Habit strength \\
$\rho_t$ & Responsiveness parameter \\
$\Delta_t$ & Time since last incentive \\
$I_{t-1}$ & Last incentive magnitude \\
\end{tabular}

\section*{Hazard Model Parameters}
\begin{tabular}{ll}
$\beta_0$ & Baseline hazard coefficient \\
$\beta_r$ & Recency coefficient \\
$\beta_h$ & Habit coefficient \\
$\beta_\rho$ & Responsiveness coefficient \\
$\beta_I$ & Incentive coefficient \\
$\lambda$ & Incentive decay rate \\
\end{tabular}

\section*{CMDP Components}
\begin{tabular}{ll}
$S$ & State space \\
A & Action space \\
$P(s'|s,a)$ & Transition probability \\
$r(s,a)$ & Reward function \\
$c(s,a)$ & Cost function \\
$B$ & Budget \\
$\gamma$ & Discount factor \\
$\lambda$ & Lagrange multiplier \\
\end{tabular}
